# The interpreter build: the game's whole program runs in psprism's interpreter, loaded from the
# player's disc as it starts, so the build holds none of the game's code. The Android app is built
# this way (the library main, which SDL's activity loads, next to SDL's own); on Linux it builds
# the executable mhfu-interpreter.
#
#   cmake -S recomp/interpreter -B build/interpreter -DPSPRISM_DIR=<psprism, patched>
#   (Android, through Gradle: also -DSDL2_SOURCE_DIR=<SDL2's source> -DFFMPEG_DIR=<FFmpeg, built
#   by android/prepare.sh>)
#
# PSPRISM_DIR is a psprism checkout at play.sh's PSPRISM_COMMIT with recomp/psprism-mhfu.patch
# applied (it needn't be built). glue/ is psprism's glue for the game's program without the
# translated functions (tools/gen_interpreter_glue.py): addresses and names only.
cmake_minimum_required(VERSION 3.22)
project(mhfu_interpreter C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PSPRISM_DIR "" CACHE PATH "psprism checkout with recomp/psprism-mhfu.patch applied")
if(NOT EXISTS "${PSPRISM_DIR}/refract/src/runtime.cpp")
  message(FATAL_ERROR "PSPRISM_DIR must be a psprism checkout with recomp/psprism-mhfu.patch applied")
endif()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(REFRACT ${PSPRISM_DIR}/refract)
set(AEMU ${REFRACT}/third_party/aemu_postoffice)

# The game's layout by the names in re/, and the build's revision, which the launcher shows.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CONFIG ${CMAKE_CURRENT_BINARY_DIR}/config)
add_custom_command(
  OUTPUT ${CONFIG}/game_layout.hpp
  COMMAND ${Python3_EXECUTABLE} -I ${REPO}/tools/gen_layout.py ${REPO} ${CONFIG}/game_layout.hpp
  DEPENDS ${REPO}/tools/gen_layout.py ${REPO}/re/functions.tsv ${REPO}/re/globals.tsv ${REPO}/re/structs.tsv)
execute_process(
  COMMAND git -C ${REPO} describe --always --dirty
  OUTPUT_VARIABLE REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT REVISION)
  set(REVISION unknown)
endif()
file(CONFIGURE OUTPUT ${CONFIG}/launcher_config.hpp CONTENT
  "// Generated by recomp/interpreter/CMakeLists.txt.\n#pragma once\nnamespace mhfu {\ninline constexpr const char* build_revision = \"@REVISION@\";\n}\n")

if(ANDROID)
  set(SDL2_SOURCE_DIR "" CACHE PATH "SDL2's source")
  add_subdirectory(${SDL2_SOURCE_DIR} SDL2)
  add_library(main SHARED)
  set(GAME main)
  target_link_libraries(main PRIVATE SDL2 GLESv3 EGL log android)
  # FFmpeg decodes the game's videos; without it they play black, with sound.
  set(FFMPEG_DIR "" CACHE PATH "FFmpeg for Android (avcodec, swscale, avutil), static")
  if(EXISTS "${FFMPEG_DIR}/lib/libavcodec.a")
    target_compile_definitions(main PRIVATE REFRACT_HAS_FFMPEG=1)
    target_include_directories(main PRIVATE ${FFMPEG_DIR}/include)
    target_link_libraries(main PRIVATE ${FFMPEG_DIR}/lib/libavcodec.a ${FFMPEG_DIR}/lib/libswscale.a
                          ${FFMPEG_DIR}/lib/libavutil.a m)
  endif()
else()
  find_package(SDL2 REQUIRED)
  add_executable(mhfu-interpreter)
  set(GAME mhfu-interpreter)
  target_link_libraries(mhfu-interpreter PRIVATE SDL2::SDL2 GLESv2)
  # FFmpeg decodes the game's videos; without it they play black, with sound.
  find_package(PkgConfig)
  if(PkgConfig_FOUND)
    pkg_check_modules(FFMPEG IMPORTED_TARGET libavcodec libavformat libavutil libswscale)
  endif()
  if(FFMPEG_FOUND)
    target_compile_definitions(mhfu-interpreter PRIVATE REFRACT_HAS_FFMPEG=1)
    target_link_libraries(mhfu-interpreter PRIVATE PkgConfig::FFMPEG)
  endif()
endif()

file(GLOB AT3_SOURCES ${REFRACT}/third_party/at3_standalone/*.cpp)
file(GLOB FEATURE_SOURCES ${REPO}/recomp/src/features/*.cpp)
target_sources(${GAME} PRIVATE
  ${CONFIG}/game_layout.hpp
  glue/dispatch.cpp
  glue/platform/platform.cpp
  ${REFRACT}/src/runtime.cpp
  ${REFRACT}/src/font/font_engine.cpp
  ${REFRACT}/src/utility_data.cpp
  ${REFRACT}/src/psp_sdk_bridge.cpp
  ${REFRACT}/src/host/audio_engine.cpp
  ${REFRACT}/src/host/common.cpp
  ${REFRACT}/src/stubs/mpeg/media_engine.cpp
  ${REFRACT}/src/host/sdl_gles.cpp
  ${REFRACT}/src/host/gpu_gl.cpp
  ${REFRACT}/src/host/gpu_vulkan.cpp
  ${AEMU}/client/delay_impl_cpp.cpp
  ${AEMU}/client/mutex_impl_cpp.cpp
  ${AT3_SOURCES}
  ${REPO}/recomp/src/launcher.cpp
  ${REPO}/recomp/src/disc.cpp
  ${REPO}/recomp/src/camera.cpp
  ${REPO}/recomp/src/fps60.cpp
  ${FEATURE_SOURCES})
target_include_directories(${GAME} PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/glue ${PSPRISM_DIR}/include ${REFRACT}/include ${REFRACT}/include/pspsdk
  ${REFRACT}/src ${REFRACT}/third_party/at3_standalone ${CONFIG})
target_compile_definitions(${GAME} PRIVATE MHFU_HAS_SDL)
# Vulkan (refract's Vulkan backend, the default where the device has it): its headers are all it
# needs, as SDL loads Vulkan. The NDK has them; on Linux, the Vulkan development package.
include(CheckIncludeFileCXX)
if(ANDROID)
  set(HAVE_VULKAN_HEADERS ON)
else()
  check_include_file_cxx(vulkan/vulkan.h HAVE_VULKAN_HEADERS)
endif()
if(HAVE_VULKAN_HEADERS)
  target_compile_definitions(${GAME} PRIVATE REFRACT_HAS_VULKAN=1)
else()
  message(STATUS "Vulkan headers not found: OpenGL ES only")
endif()
# Always optimized: the interpreter runs the whole game.
target_compile_options(${GAME} PRIVATE -O2 -w)

# The client for ad hoc servers (aemu_postoffice), for multiplayer: C, and its session with the
# server, host code compiled without the PSP SDK's headers (they stand in for the system's).
add_library(aemu OBJECT
  ${AEMU}/client/postoffice.c ${AEMU}/client/list.c ${AEMU}/client/sock_impl_linux.c
  ${AEMU}/client/postoffice_mem_stdc.c ${AEMU}/client/enet.c
  ${AEMU}/adhocctl/client/adhocctl.c ${AEMU}/adhocctl/client/adhocctl_mem.c
  ${REFRACT}/src/net/adhoc_control.cpp)
target_include_directories(aemu PRIVATE ${REFRACT}/include)
target_compile_options(aemu PRIVATE -O2 -w)
set_target_properties(aemu PROPERTIES POSITION_INDEPENDENT_CODE ON)

find_package(Threads REQUIRED)
target_link_libraries(${GAME} PRIVATE aemu Threads::Threads ${CMAKE_DL_LIBS})
